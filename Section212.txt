FORM USER_COMMAND USING R_UCOMM LIKE SY-UCOMM
                  RS_SELFIELD TYPE SLIS_SELFIELD.

    CASE R_UCOMM.
    WHEN '&IC1'.
       CLEAR : GS_LIST.
       READ TABLE GT_LIST INTO GS_LIST INDEX RS_SELFIELD-TABINDEX.
       IF GS_LIST-EMPLOYEE_ID IS NOT INITIAL.
          PERFORM CALL_SF.
      ENDIF.

    ENDCASE.

   CLEAR : GS_LIST, SY-UCOMM.

ENDFORM.
